// ─── Hooks ───────────────────────────────────────────────────────────────────

function useOutsideClick(ref: React.RefObject<HTMLElement | null>, handler: () => void) {
  useEffect(() => {
    const listener = (e: MouseEvent) => {
      if (!ref.current || ref.current.contains(e.target as Node)) return;
      handler();
    };
    document.addEventListener("mousedown", listener);
    return () => document.removeEventListener("mousedown", listener);
  }, [ref, handler]);
}

          {/* Results header */}
          <div className="flex items-center justify-between mb-6">
            <div>
              <h2 className="text-[20px] font-bold text-[#1e293b]">
                {activeSubcategoryData?.label ?? "Tous les groupes"}
              </h2>
              <p className="text-[13px] text-[#64748b] mt-0.5">
                {filtered.length} produit{filtered.length !== 1 ? "s" : ""}
                {activeSubcategoryData?.description ? ` — ${activeSubcategoryData.description}` : ""}
              </p>
            </div>
            {(activeBrand !== "Tous" || activeCategory !== "all") && (
              <button
                onClick={() => {
                  setActiveCategory("all");
                  setActiveBrand("Tous");
                  setActivePower("all");
                }}
                className="flex items-center gap-1.5 text-[12px] font-semibold text-[#64748b] hover:text-[#1e293b] transition-colors cursor-pointer"
              >
                <X size={13} weight="bold" />
                Réinitialiser
              </button>
            )}
          </div>

          {/* Product Grid */}
          {filtered.length === 0 ? (
            <div className="bg-white rounded-[1.5rem] border border-[#e2e8f0] p-20 text-center">
              <div className="w-16 h-16 rounded-2xl bg-[#f1f5f9] flex items-center justify-center mx-auto mb-5">
                <MagnifyingGlass size={28} weight="thin" className="text-[#94a3b8]" />
              </div>
              <p className="text-[16px] font-bold text-[#334155] mb-2">Aucun produit trouvé</p>
              <p className="text-[13px] text-[#94a3b8] mb-6">Essayez d'autres filtres ou explorez toute la gamme</p>
              <button
                onClick={() => {
                  setActiveCategory("all");
                  setActiveBrand("Tous");
                }}
                className="inline-flex items-center gap-2 bg-primary text-white text-[13px] font-bold px-5 py-2.5 rounded-full hover:bg-primary/90 transition-colors cursor-pointer"
              >
                Voir tous les produits <ArrowRight size={13} weight="bold" />
              </button>
            </div>
          ) : (
            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-5 lg:gap-6">
              {filtered.map((product) => (
                <ProductCard
                  key={product.id}
                  product={product}
                  liked={likedProducts.has(product.id)}
                  onToggleLike={toggleLike}
                />
              ))}
            </div>
          )}

          {/* Pagination */}
          {filtered.length > 0 && (
            <div className="mt-12 flex items-center justify-between">
              <div className="flex items-center gap-2">
                <div className="w-5 h-5 rounded-full border-2 border-[#1e293b] flex items-center justify-center">
                  <div className="w-2 h-2 rounded-full bg-[#1e293b]" />
                </div>
                {[1, 2, 3].map((i) => (
                  <div key={i} className="w-1.5 h-1.5 rounded-full bg-[#cbd5e1]" />
                ))}
              </div>
              <div className="flex items-center gap-2.5">
                <button className="w-11 h-10 rounded-full flex items-center justify-center text-[#94a3b8] border border-[#e2e8f0] bg-white hover:text-primary hover:border-primary/40 transition-all duration-150 cursor-pointer shadow-sm">
                  <ArrowRight size={16} weight="bold" className="rotate-180" />
                </button>
                <button className="w-11 h-10 rounded-full bg-primary flex items-center justify-center text-white hover:bg-primary/90 transition-colors cursor-pointer shadow-md shadow-primary/25">
                  <ArrowRight size={16} weight="bold" />
                </button>
              </div>
            </div>
          )}
        </div>
      </section>

      {/* ── Trust Features ───────────────────────────────────────────── */}
      <section className="bg-white border-y border-[#e2e8f0] px-6 py-12 lg:px-10 lg:py-16">
        <div className="mx-auto max-w-300">
          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-8">
            {features.map((f) => {
              const Icon = f.icon;
              return (
                <div key={f.title} className="flex flex-col gap-4">
                  <div className="flex items-center gap-3">
                    <div className="w-11 h-11 rounded-xl bg-primary/6 flex items-center justify-center shrink-0">
                      <Icon size={20} weight="bold" className="text-primary" />
                    </div>
                    <span className="text-[22px] font-bold text-[#1e293b] tabular-nums">{f.stat}</span>
                  </div>
                  <div>
                    <h3 className="text-[14px] font-bold text-[#1e293b] mb-1">{f.title}</h3>
                    <p className="text-[13px] text-[#64748b] leading-relaxed">{f.desc}</p>
                  </div>
                </div>
              );
            })}
          </div>
        </div>
      </section>

      {/* ── CTA Banner ───────────────────────────────────────────────── */}
      <section className="px-6 py-12 lg:px-10 lg:py-16">
        <div className="mx-auto max-w-300">
          <div className="bg-primary rounded-[2rem] p-10 lg:p-14 flex flex-col lg:flex-row items-start lg:items-center justify-between gap-8 relative overflow-hidden">
            {/* BG decoration */}
            <div className="absolute -right-16 -top-16 w-80 h-80 bg-white/4 rounded-full" />
            <div className="absolute right-32 -bottom-20 w-56 h-56 bg-[#f2a74c]/8 rounded-full" />
            <div
              className="absolute inset-0 opacity-[0.025]"
              style={{
                backgroundImage:
                  "repeating-linear-gradient(45deg, white 0, white 1px, transparent 0, transparent 50%)",
                backgroundSize: "20px 20px",
              }}
            />

            <div className="relative z-10">
              <span className="text-[#f2a74c] text-[11px] font-bold uppercase tracking-widest mb-3 block">
                Projets industriels & achat en gros
              </span>
              <h2 className="text-[28px] lg:text-[32px] font-bold text-white mb-3 leading-tight">
                Besoin d'un devis personnalisé ?
              </h2>
              <p className="text-[14px] text-white/55 leading-relaxed max-w-md">
                Notre équipe commerciale vous propose des tarifs négociés, un accompagnement sur mesure et des solutions clé en main.
              </p>
            </div>

            <div className="relative z-10 flex flex-col sm:flex-row lg:flex-col xl:flex-row gap-3 shrink-0">
              <Link
                href="/achat-en-gros"
                className="flex items-center justify-center gap-2 bg-[#f2a74c] text-white text-[13px] font-bold px-7 py-3.5 rounded-full hover:bg-[#e09640] transition-colors shadow-lg cursor-pointer"
              >
                Demander un devis
                <ArrowRight size={14} weight="bold" />
              </Link>
              <Link
                href="/contact"
                className="flex items-center justify-center gap-2 bg-white/10 text-white text-[13px] font-bold px-7 py-3.5 rounded-full border border-white/15 hover:bg-white/18 transition-colors cursor-pointer"
              >
                Nous contacter
              </Link>
            </div>
          </div>
        </div>
      </section>

    </main>
  );
}
